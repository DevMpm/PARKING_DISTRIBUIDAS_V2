\section{Zonas}
\subsection{Crear Zona}

\casodeprueba{CP1}
{Crear una zona con datos correctos.}
{Token JWT con roles suficientes}
{JSON con datos válidos}
{\begin{enumerate} [leftmargin=*, nosep]
    \item Código de respuesta 201
    \item Respuesta DTO con datos correctos.
    \item Verificar formato del código de la zona.
    \item Buscar zona y encontrar que existe.
\end{enumerate}}

\casodeprueba{CP2}
{(Caso adicional, zona con datos correctos y tipo en minúsculas)}
{Token JWT con roles suficientes}
{JSON con datos válidos}
{\begin{enumerate} [leftmargin=*, nosep]
    \item Se verifica el cumplimiento de la operación.
\end{enumerate}}

\casodeprueba{CP3}
{Crear zona con datos incorrectos.}
{Token JWT con roles suficientes}
{JSON con datos inválidos (4 casos separados. CP3.1-CP3.4)}
{\begin{enumerate} [leftmargin=*, nosep]
    \item Nombre inválido
    \item Nombre duplicado
    \item Tipo inválido
    \item Capacidad invalida
    \item Códigos de respuesta 400 o 409 en el caso del duplicado
    \item Mensajes de error
    \item Buscar zona y no encontrar.
\end{enumerate}}

\casodeprueba{CP4}
{Obtener zonas normal}
{Token JWT con roles suficientes y 4 zonas creadas.}
{Ninguna}
{\begin{enumerate} [leftmargin=*, nosep]
    \item Código de respuesta 201
    \item Cantidad de registros en el array.
\end{enumerate}}

\casodeprueba{CP4.1}
{Obtener zonas desocupadas si hay}
{Token JWT con roles suficientes y 4 zonas creadas cada una con 1 espacio.}
{Ninguna}
{\begin{enumerate} [leftmargin=*, nosep]
    \item Código de respuesta 200
    \item Cantidad de registros en el array.
    \item Verificar los espacios desocupados en las zonas de respuesta.
\end{enumerate}}

\casodeprueba{CP4.2}
{Obtener zonas desocupadas si no hay}
{Token JWT con roles suficientes y 4 zonas creadas cada una con 1 espacio.}
{Ninguna}
{\begin{enumerate} [leftmargin=*, nosep]
    \item Código de respuesta 404
    \item Cantidad de registros en el array.
\end{enumerate}}

\casodeprueba{CP5}
{CP9. Obtener zonas por tipo}
{Token JWT con roles suficientes y 1 zona creada por cada tipo.}
{Ninguna}
{\begin{enumerate} [leftmargin=*, nosep]
    \item Tipo de zona como parámetro
    \item Cantidad de resultados de cada búsqueda
    \item Código de la zona obtenida (Debe coincidir con el tipo buscado)
\end{enumerate}}

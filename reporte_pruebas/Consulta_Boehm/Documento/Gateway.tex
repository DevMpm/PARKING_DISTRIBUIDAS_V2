\section{Gateway}
\subsection{Módulo de Entradas y Salidas (Gateway)}

\subsection{Consultar Estado de Vehículo por Placa (GET /gateway/vehiculos/:placa)}

\casodeprueba{CP1.1}
{Consultar vehículo por placa existente ok}
{Token JWT con roles suficientes y un vehículo registrado en el sistema.}
{Placa válida existente como parámetro en la URL (GET /gateway/vehiculos/ABC-1234).}
{\begin{enumerate} [leftmargin=*, nosep]
    \item Código de respuesta 200
    \item DTO de respuesta con { registrado: true, placa, tipo, clasificacion, estado, activo }.
\end{enumerate}}

\casodeprueba{CP1.2}
{Consultar vehículo por placa no existente (sin resultados)}
{Token JWT con roles suficientes.}
{Placa no registrada en el sistema (GET /gateway/vehiculos/XYZ-9999).}
{\begin{enumerate} [leftmargin=*, nosep]
    \item Código de respuesta 200 (diseño de preflight que no lanza 404).
    \item DTO de respuesta con { registrado: false }.
\end{enumerate}}

\casodeprueba{CP2.1}
{Autorizar ingreso ok}
{Token JWT con roles suficientes, vehículo registrado con estado FUERA, y cupo disponible en ModuloZonas.}
{JSON con placa válida ({ 'placa': 'ABC-1234' }) en POST /gateway/ingresos.}
{\begin{enumerate} [leftmargin=*, nosep]
    \item Código de respuesta 200
    \item Respuesta DTO con { autorizado: true, vehiculoId, placa, tipo, clasificacion, fechaIngreso }.
    \item Buscar estado del vehículo para confirmar que ahora se encuentra en estado DENTRO.
\end{enumerate}}

\casodeprueba{CP2.2}
{Autorizar ingreso - Vehículo con ingreso activo (Conflicto / Duplicado)}
{Token JWT con roles suficientes y un vehículo que ya posee un ingreso activo (estado DENTRO).}
{JSON con la placa del vehículo activo ({ 'placa': 'ABC-1234' }) en POST /gateway/ingresos.}
{\begin{enumerate} [leftmargin=*, nosep]
    \item Código de respuesta 409
    \item Mensaje de error indicando que ya tiene un ingreso activo o no hay cupo disponible.
\end{enumerate}}

\casodeprueba{CP2.3}
{Autorizar ingreso - Vehículo no encontrado}

\end{enumerate}}

\casodeprueba{CP2.4}

\casodeprueba{CP3.1}
{Registrar vehículo e ingresar ok}
{Token JWT con roles suficientes, placa no registrada previamente y cupo disponible en ModuloZonas.}

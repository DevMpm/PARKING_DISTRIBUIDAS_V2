\section{Roles}
\subsection{Módulo de Roles (/roles)}

\subsection{Crear Rol (POST /roles)}

\casodeprueba{CP1.1}
{Crear rol ok}
{Token JWT con rol ADMIN o ROOT y permiso ROLES\_CREATE.}
{JSON con datos válidos { 'name': 'SUPERVISOR', 'description': 'Rol de supervisión de estacionamiento' }.}
{\begin{enumerate} [leftmargin=*, nosep]
    \item Código de respuesta 201
    \item Respuesta con datos correctos del rol creado.
    \item Buscar por ID (GET /roles/:id) o por nombre (GET /roles/name/:name) para confirmar creación.
\end{enumerate}}

\casodeprueba{CP1.2}
{Crear rol duplicado}
{Token JWT con permisos suficientes y un rol ya existente con nombre 'ADMIN'.}
{JSON con el mismo nombre existente { 'name': 'ADMIN', 'description': 'Intento duplicado' }.}
{\begin{enumerate} [leftmargin=*, nosep]
    \item Código de respuesta 409
    \item Mensaje de error indicando que ya existe un rol con ese nombre.
\end{enumerate}}

\casodeprueba{CP1.3.1}
{Crear rol - Nombre vacío o ausente}
{Token JWT con permisos suficientes.}
{JSON con name vacío ('') o ausente.}
{\begin{enumerate} [leftmargin=*, nosep]
    \item Código de respuesta 400
    \item Mensaje de error indicando que el nombre del rol es obligatorio.
\end{enumerate}}

\casodeprueba{CP1.3.2}
{Crear rol - Nombre no es string}
{Token JWT con permisos suficientes.}
{JSON con name numérico (123).}
{\begin{enumerate} [leftmargin=*, nosep]
    \item Código de respuesta 400
    \item Mensaje de error indicando que el nombre del rol debe ser texto.
\end{enumerate}}

\casodeprueba{CP1.3.3}
{Crear rol - Descripción excede longitud máxima}
{Token JWT con permisos suficientes.}
{JSON con description de más de 255 caracteres.}
{\begin{enumerate} [leftmargin=*, nosep]
    \item Código de respuesta 400
    \item Mensaje de error indicando que la descripción no puede tener más de 255 caracteres.
\end{enumerate}}
